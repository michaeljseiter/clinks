import (
    "errors"
    "fmt"
    "net/url"
)

// Validate checks if the profile is valid
func (p *Profile) Validate() error {
    if p.Name == "" {
        return errors.New("name is required and must not be empty")
    }

    if len(p.Links) == 0 {
        return errors.New("links must be a non-empty array")
    }

    for i, link := range p.Links {
        if err := link.Validate(); err != nil {
            return fmt.Errorf("link[%d]: %w", i, err)
        }
    }

    return nil
}

// Validate checks if a link is valid
func (l *Link) Validate() error {
    if l.Label == "" {
        return errors.New("label is required and must not be empty")
    }

    if l.URL == "" {
        return errors.New("url is required and must not be empty")
    }

    parsedURL, err := url.Parse(l.URL)
    if err != nil {
        return fmt.Errorf("invalid url: %w", err)
    }

    if !parsedURL.IsAbs() {
        return errors.New("url must be absolute (include scheme like https://)")
    }

    // Validate scheme (url.Parse is very permissive)
    scheme := parsedURL.Scheme
    if scheme != "http" && scheme != "https" && scheme != "mailto" {
        return fmt.Errorf("url scheme must be http, https, or mailto (got %q)", scheme)
    }

    // For http/https, ensure host is present
    if (scheme == "http" || scheme == "https") && parsedURL.Host == "" {
        return errors.New("http/https url must have a host")
    }

    return nil
}
#+end_src

package profile

import (
    "encoding/json"
    "errors"
    "fmt"
    "os"
    "path/filepath"
)

// FindConfigFile searches for clinks.json in standard locations
func FindConfigFile() (string, error) {
    // Try current working directory first
    if _, err := os.Stat("clinks.json"); err == nil {
        return "clinks.json", nil
    }

    // Try config directory
    configDir, err := os.UserConfigDir()
    if err != nil {
        return "", fmt.Errorf("could not determine config directory: %w", err)
    }

    configPath := filepath.Join(configDir, "clinks", "clinks.json")
    if _, err := os.Stat(configPath); err == nil {
        return configPath, nil
    }

    return "", errors.New("clinks.json not found")
}

// LoadProfile reads and validates a profile from a file
func LoadProfile(path string) (*Profile, error) {
    data, err := os.ReadFile(path)
    if err != nil {
        return nil, fmt.Errorf("could not read file: %w", err)
    }

    var profile Profile
    if err := json.Unmarshal(data, &profile); err != nil {
        return nil, fmt.Errorf("invalid JSON: %w", err)
    }

    if err := profile.Validate(); err != nil {
        return nil, fmt.Errorf("validation failed: %w", err)
    }

    return &profile, nil
}

package profile

import (
    "fmt"
    "strings"
)

// RenderText renders a profile as plain text
func RenderText(p *Profile) string {
    var sb strings.Builder

    // Print name
    sb.WriteString(p.Name)
    sb.WriteString("\n")

    // Print divider (matching name length, max 50)
    // Note: len() counts bytes, not Unicode runes. Good enough for v0.1
    dividerLen := len(p.Name)
    if dividerLen > 50 {
        dividerLen = 50
    }
    sb.WriteString(strings.Repeat("─", dividerLen))
    sb.WriteString("\n")

    // Find max label length for alignment
    maxLabelLen := 0
    for _, link := range p.Links {
        if len(link.Label) > maxLabelLen {
            maxLabelLen = len(link.Label)
        }
    }

    // Print links
    for _, link := range p.Links {
        // Left-align label in column
        sb.WriteString(link.Label)
        sb.WriteString(strings.Repeat(" ", maxLabelLen-len(link.Label)+2))
        sb.WriteString(link.URL)
        sb.WriteString("\n")
    }

    return sb.String()
}

package main

import (
    "fmt"
    "os"
    "path/filepath"

    "github.com/YOURUSERNAME/clinks/internal/profile"
)

func main() {
    if err := run(); err != nil {
        fmt.Fprintf(os.Stderr, "Error: %v\n", err)
        os.Exit(1)
    }
}

func run() error {
    // Find config file
    configPath, err := profile.FindConfigFile()
    if err != nil {
        return formatNotFoundError()
    }

    // Load and validate profile
    prof, err := profile.LoadProfile(configPath)
    if err != nil {
        return fmt.Errorf("failed to load profile: %w", err)
    }

    // Render and print
    output := profile.RenderText(prof)
    fmt.Print(output)

    return nil
}

func formatNotFoundError() error {
    // Compute actual config path for this OS
    configDir, err := os.UserConfigDir()
    configPath := "~/.config/clinks/clinks.json" // fallback
    if err == nil {
        configPath = filepath.Join(configDir, "clinks", "clinks.json")
    }

    return fmt.Errorf(`clinks.json not found

Looked in:
  ./clinks.json
  %s

Create a clinks.json file with your profile:
  {
    "name": "Your Name",
    "links": [
      {"label": "github", "url": "https://github.com/username"}
    ]
  }`, configPath)
}

# clinks

A minimal CLI tool for personal link profiles.

## Installation

```bash
go install github.com/YOURUSERNAME/clinks@latest
```

## Usage

Create a `clinks.json` file:

```json
{
  "name": "Your Name",
  "links": [
    {"label": "github", "url": "https://github.com/username"},
    {"label": "blog", "url": "https://example.com"}
  ]
}
```

Run:

```bash
clinks
```

Output:

```
Your Name
─────────
github  https://github.com/username
blog    https://example.com
```

## File Locations

clinks looks for `clinks.json` in:
1. Current working directory
2. `~/.config/clinks/clinks.json`

## Philosophy

- No accounts, no tracking
- You own your domain
- Convention over configuration

* Key Design Decisions
** Package Naming
Use =package profile= instead of =package internal= inside the =internal/= directory. This is more idiomatic and less confusing when importing.

** URL Validation
Go's =url.Parse()= is very permissive (e.g., ="not-a-url"= can sometimes parse successfully). We add explicit checks for:
- Allowed schemes: http, https, mailto
- Host must be present for http/https URLs

This ensures reliable validation and stable tests.

** Cross-Platform Paths
Error messages compute the actual config path using =os.UserConfigDir()= rather than hardcoding =~/.config=. This shows the correct path on macOS (=~/Library/Application Support=) and Windows (=%AppData%=).

** Divider Length
The divider uses =len(p.Name)= which counts bytes, not Unicode runes. This is acceptable for v0.1. A proper Unicode-aware implementation would use =utf8.RuneCountInString()=, but that's overkill for now.
